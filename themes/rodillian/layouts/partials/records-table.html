{{/* records-table.html Renders a TablePress-style records table. Input: a record object with: - .header: array of column headers - .rows: array of rows (first 3 = distance 1, next 3 = distance 2, etc.) Format: each distance takes 3 rows: - row 1: distance name (col 0) + athlete names - row 2: empty (col 0) + times - row 3: empty (col 0) + dates */}}
{{ range .header }} {{ end }} {{ $rows := .rows }} {{ $numRecords := div (len $rows) 3 }} {{ range $i := (seq 0 (sub $numRecords 1)) }} {{ $nameRow := index $rows (mul $i 3) }} {{ $timeRow := index $rows (add 1 (mul $i 3)) }} {{ $dateRow := index $rows (add 2 (mul $i 3)) }} {{ range $j, $name := $nameRow }} {{ if gt $j 0 }}{{ end }} {{ end }} {{ range $j, $time := $timeRow }} {{ if gt $j 0 }}{{ end }} {{ end }} {{ range $j, $date := $dateRow }} {{ if gt $j 0 }}{{ end }} {{ end }} {{ end }}
{{ . }}
{{ index $nameRow 0 }}{{ $name }}
{{ $time }}
{{ $date }}
